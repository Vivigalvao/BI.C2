CREATE TABLE dw.dim_tempo (
    sk_tempo INTEGER PRIMARY KEY,
    data DATE,
    dia SMALLINT,
    mes SMALLINT,
    ano SMALLINT,
    trimestre SMALLINT,
    nome_mes VARCHAR(15),
    dia_semana VARCHAR(15),
    ano_mes CHAR(7),
    eh_fim_de_semana BOOLEAN,
    semana_epidemiologica SMALLINT
);
